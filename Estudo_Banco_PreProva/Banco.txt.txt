CREATE DATABASE BANCO;
USE BANCO;


CREATE TABLE CLIENTE (
     id int AUTO_INCREMENT PRIMARY KEY,
     nome varchar(50) NOT NULL,
     idade int NOT NULL,
     saldo float NOT NULL
);

CREATE TABLE FUNCIONARIO (
     id int AUTO_INCREMENT PRIMARY KEY,
     nome varchar(50) NOT NULL,
     idade int NOT NULL,
     clientesAtendidos int NOT NULL
);
